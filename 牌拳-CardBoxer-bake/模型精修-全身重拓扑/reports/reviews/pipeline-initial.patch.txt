+def point_surface_distance(point, triangles):
+    """Minimum Euclidean distance to triangles, including interior projections."""
+    if len(triangles)==0: return float("inf")
+    a,b,c=triangles[:,0],triangles[:,1],triangles[:,2]
+    best=np.full(len(a),np.inf)
+    for start,end in ((a,b),(b,c),(c,a)):

+        best=np.minimum(best,np.sum((point-(start+t[:,None]*e))**2,axis=1))
+    u=b-a;v=c-a;n=np.cross(u,v);n2=np.sum(n*n,axis=1)

+    projection=point-dist[:,None]*n;q=projection-a
+    uu=np.sum(u*u,axis=1);uv=np.sum(u*v,axis=1);vv=np.sum(v*v,axis=1)
+    qu=np.sum(q*u,axis=1);qv=np.sum(q*v,axis=1);den=uu*vv-uv*uv

+    inside=(s>=0)&(t>=0)&(s+t<=1)&(n2>1e-20)
+    best[inside]=np.minimum(best[inside],dist[inside]**2*n2[inside])

+def nearest_error(reference,points,tolerance=1e-5):
+    """Spatial hash independent of exporter ordering; search neighboring cells."""
+    cells={}
+    for i,key in enumerate(np.floor(reference/tolerance).astype(np.int64)):
+        cells.setdefault(tuple(key),[]).append(i)
+    offsets=list(itertools.product((-1,0,1),repeat=3));worst=0.
+    for point,key in zip(points,np.floor(points/tolerance).astype(np.int64)):
+        candidates=[]
+        for delta in offsets:candidates.extend(cells.get(tuple(key+delta),()))
+        if not candidates:return float("inf")
+        distance=float(np.linalg.norm(reference[candidates]-point,axis=1).min())
+        worst=max(worst,distance)
+    return worst
