          前者遍历两版表全量 284 帧得 D_static = 1.870037° @wave:34。

  export PATH="/c/Users/tutic/.workbuddy/binaries/PortableGit/versions/1.2.0/usr/bin:/c/Users/tutic/.workbuddy/binaries/PortableGit/versions/1.2.0/bin:$PATH"
  sha256sum \
    "D:/Projects/CardBoxer/CardBoxer/data/"{card_motion,motion_input,motion_input_foe}.json \
    "D:/Projects/CardBoxer/CardBoxer-bake/native-plateau-p4/orig/pre-p4/"{card_motion,motion_input,motion_input_foe}.json
